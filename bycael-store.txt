function toast(msg){const t=$('toast');t.textContent=msg;t.classList.add('show');clearTimeout(toastTimer);toastTimer=setTimeout(()=>t.classList.remove('show'),2600)}
function escapeHTML(s){return String(s==null?'':s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]))}
function safeImage(url){return /^https?:\/\//i.test(String(url||''))?String(url):''}
function formatCategory(s){return String(s||'Everyday').trim()||'Everyday'}
$('year').textContent=new Date().getFullYear();
if(isAdmin){$('storeApp').classList.add('hidden');$('adminApp').classList.remove('hidden');initAdmin();return}
function renderCategories(){
 const cats=Array.from(new Set(products.map(p=>formatCategory(p.category)))).sort((a,b)=>a.localeCompare(b));
 $('categoryChips').innerHTML='<button class="chip '+(category==='all'?'active':'')+'" data-cat="all">Everything</button>'+cats.map(c=>'<button class="chip '+(category===c?'active':'')+'" data-cat="'+escapeHTML(c)+'">'+escapeHTML(c)+'</button>').join('');
 $('categoryChips').querySelectorAll('button').forEach(b=>b.addEventListener('click',()=>{category=b.dataset.cat;renderCategories();renderProducts()}));
}
function filteredProducts(){const q=$('searchInput').value.trim().toLowerCase();return products.filter(p=>(category==='all'||formatCategory(p.category)===category)&&(!q||[p.name,p.description,p.category].join(' ').toLowerCase().includes(q)))}
function renderProducts(){
 const list=filteredProducts();
 if(!list.length){$('productGrid').innerHTML='<div class="empty"><strong style="display:block;color:#34483b;margin-bottom:8px">'+(products.length?'No matches just yet.':'Your next favorite is on its way.')+'</strong>'+(products.length?'Try another search or category.':'Check back soon for our first curated finds.')+'</div>';return}
 $('productGrid').innerHTML=list.map(p=>{const img=safeImage(p.image_url);return '<article class="product"><div class="product-img">'+(img?'<img loading="lazy" src="'+escapeHTML(img)+'" alt="'+escapeHTML(p.name)+'" onerror="this.style.display=\'none\';this.nextElementSibling.style.display=\'grid\'"><div class="product-placeholder" style="display:none">b.</div>':'<div class="product-placeholder">b.</div>')+'<span class="tag">'+escapeHTML(formatCategory(p.category))+'</span><button class="quickadd" data-add="'+p.id+'" aria-label="Add '+escapeHTML(p.name)+' to bag">+</button></div><div class="product-info"><div class="product-cat">THE BYCAEL EDIT</div><h3 class="product-name">'+escapeHTML(p.name)+'</h3><div class="product-desc">'+escapeHTML(p.description||'A thoughtful pick for your everyday.')+'</div><div class="product-bottom"><span class="price">'+money(p.price)+'</span><button class="small-add" data-add="'+p.id+'">Add to bag +</button></div></div></article>'}).join('');
 $('productGrid').querySelectorAll('[data-add]').forEach(b=>b.addEventListener('click',()=>addToCart(b.dataset.add)));
}
async function loadProducts(){
 $('productGrid').innerHTML='<div class="empty">Finding the good stuff…</div>';
 const {data,error}=await db.from('products').select('id,name,description,price,image_url,category,whatsapp_number,is_active,created_at').eq('is_active',true).order('created_at',{ascending:false});
 if(error){$('productGrid').innerHTML='<div class="empty">We could not load the catalog right now. Please refresh the page in a moment.</div>';console.error(error);return}
 products=data||[];renderCategories();renderProducts();
}
$('searchInput').addEventListener('input',renderProducts);
function addToCart(id){const p=products.find(x=>x.id===id);if(!p)return;const row=cart.find(x=>x.id===id);if(row)row.qty++;else cart.push({id:p.id,name:p.name,price:Number(p.price)||0,image_url:p.image_url,whatsapp_number:p.whatsapp_number,qty:1});renderCart();toast('Added to your bag ✳')}
function changeQty(id,delta){const item=cart.find(x=>x.id===id);if(!item)return;item.qty+=delta;if(item.qty<1)cart=cart.filter(x=>x.id!==id);renderCart()}
function renderCart(){
 $('cartCount').textContent=cart.reduce((s,x)=>s+x.qty,0);
 $('cartItems').innerHTML=cart.length?cart.map(x=>'<div class="cartrow">'+(safeImage(x.image_url)?'<img src="'+escapeHTML(safeImage(x.image_url))+'" alt="">':'<div class="prodthumb" style="display:grid;place-items:center;font-weight:800">b.</div>')+'<div><strong>'+escapeHTML(x.name)+'</strong><small>'+money(x.price)+'</small><div class="qty"><button data-qty="'+x.id+'" data-delta="-1" aria-label="Decrease quantity">−</button><span style="font-size:11px">'+x.qty+'</span><button data-qty="'+x.id+'" data-delta="1" aria-label="Increase quantity">+</button></div></div><strong style="font-size:11px">'+money(x.price*x.qty)+'</strong></div>').join(''):'<div class="cart-empty"><div style="font-size:30px;margin-bottom:10px">♡</div>Your bag is waiting for something good.<br><span style="font-size:11px">Add a favorite to get started.</span></div>';
 $('cartTotal').textContent=money(cart.reduce((s,x)=>s+x.price*x.qty,0));
 $('cartItems').querySelectorAll('[data-qty]').forEach(b=>b.addEventListener('click',()=>changeQty(b.dataset.qty,Number(b.dataset.delta))));
 $('checkoutBtn').disabled=!cart.length;$('checkoutBtn').style.opacity=cart.length?'1':'.5';
}
$('cartOpen').addEventListener('click',()=>{$('cartPanel').classList.add('open');document.body.style.overflow='hidden'});
function closeCart(){$('cartPanel').classList.remove('open');document.body.style.overflow=''}
$('cartClose').addEventListener('click',closeCart);$('cartPanel').addEventListener('click',e=>{if(e.target===$('cartPanel'))closeCart()});
$('checkoutBtn').addEventListener('click',()=>{
 if(!cart.length)return;
 const phone=(cart.find(x=>x.whatsapp_number&&String(x.whatsapp_number).replace(/\D/g,'').length>7)||{}).whatsapp_number;
 const lines=cart.map(x=>'- '+x.name+' x'+x.qty+' — '+money(x.price*x.qty));
 const msg='Halo bycaeL! Aku mau order ya ✨\n\n'+lines.join('\n')+'\n\nSubtotal: '+money(cart.reduce((s,x)=>s+x.price*x.qty,0))+'\n\nBoleh info ongkir dan langkah selanjutnya?';
 const number=String(phone||'').replace(/\D/g,'');
 if(number.length>7)location.href='https://wa.me/'+number+'?text='+encodeURIComponent(msg);
 else {const generic=cart.find(x=>x.whatsapp_number&&String(x.whatsapp_number).replace(/\D/g,'').length>7);if(generic)location.href='https://wa.me/'+String(generic.whatsapp_number).replace(/\D/g,'')+'?text='+encodeURIComponent(msg);else{toast('WhatsApp store belum disetel. Silakan hubungi admin toko.');}}
});
$('contactBtn').addEventListener('click',()=>{const p=products.find(x=>x.whatsapp_number&&String(x.whatsapp_number).replace(/\D/g,'').length>7);if(p)location.href='https://wa.me/'+String(p.whatsapp_number).replace(/\D/g,'')+'?text='+encodeURIComponent('Halo bycaeL! Aku mau tanya tentang produk ya ✨');else toast('Nomor WhatsApp toko belum disetel oleh admin.')});
renderCart();loadProducts();
function initAdmin(){
 const show=(id,on)=>$(id).classList.toggle('hidden',!on);
 let currentUser=null;
 function authNotice(msg,warn){$('authNotice').textContent=msg;$('authNotice').classList.remove('hidden','warn');if(warn)$('authNotice').classList.add('warn')}
 function saveNotice(msg,warn){$('saveNotice').textContent=msg;$('saveNotice').classList.remove('hidden','warn');if(warn)$('saveNotice').classList.add('warn')}
 function resetForm(){ $('productForm').reset();$('productId').value='';$('productActive').checked=true;$('editorTitle').textContent='Add a product';$('saveProductBtn').textContent='Save product ↗';$('saveNotice').classList.add('hidden') }
 function setAuthView(user,admin){
  currentUser=user;show('loginPanel',!user);show('pendingPanel',!!user&&!admin);show('dashboard',!!user&&!!admin);show('logoutBtn',!!user);
  if(user&&!admin){$('pendingText').textContent='Signed in as '+user.email+'. This account does not have store-admin permission yet.'}
 }
 async function checkSession(){
  const {data}=await db.auth.getSession();const user=data.session&&data.session.user;
  if(!user){setAuthView(null,false);return}
  const {data:row,error}=await db.from('store_admins').select('user_id').eq('user_id',user.id).maybeSingle();
  if(error){setAuthView(user,false);authNotice('Could not verify admin permission. Please try signing in again.');return}
  setAuthView(user,!!row);if(row)await refreshAdminProducts();
 }
 $('authForm').addEventListener('submit',async e=>{e.preventDefault();$('loginBtn').disabled=true;$('loginBtn').textContent='Signing in…';
  const {data,error}=await db.auth.signInWithPassword({email:$('authEmail').value.trim(),password:$('authPassword').value});
  $('loginBtn').disabled=false;$('loginBtn').textContent='Sign in ↗';
  if(error){authNotice(error.message,true);return}await checkSession();
 });
 $('signupBtn').addEventListener('click',async()=>{
  const email=$('authEmail').value.trim(),password=$('authPassword').value;
  if(!email||password.length<6){authNotice('Enter a valid email and a password with at least 6 characters.',true);return}
  $('signupBtn').disabled=true;$('signupBtn').textContent='Creating…';
  const {data,error}=await db.auth.signUp({email,password});
  $('signupBtn').disabled=false;$('signupBtn').textContent='Create account';
  if(error){authNotice(error.message,true);return}
  if(data.session){authNotice('Account created. Check whether email confirmation is required. Admin access must still be approved by the owner.',false);await checkSession()}
  else authNotice('Account request created. Check your inbox for the confirmation link. After confirming, sign in. Admin permission must still be approved by the owner.',false);
 });
 async function logout(){await db.auth.signOut();setAuthView(null,false)}
 $('logoutBtn').addEventListener('click',logout);$('pendingLogout').addEventListener('click',logout);
 $('newProductBtn').addEventListener('click',()=>{resetForm();$('productEditor').scrollIntoView({behavior:'smooth',block:'start'});$('productName').focus()});
 $('resetFormBtn').addEventListener('click',resetForm);$('refreshBtn').addEventListener('click',refreshAdminProducts);
 async function refreshAdminProducts(){
  const {data,error}=await db.from('products').select('*').order('created_at',{ascending:false});
  if(error){$('adminProducts').innerHTML='<tr><td colspan="4">Could not load products. Check your admin permission.</td></tr>';return}
  adminProducts=data||[];$('statAll').textContent=adminProducts.length;$('statLive').textContent=adminProducts.filter(p=>p.is_active).length;$('statHidden').textContent=adminProducts.filter(p=>!p.is_active).length;
  $('adminProducts').innerHTML=adminProducts.length?adminProducts.map(p=>'<tr><td>'+(safeImage(p.image_url)?'<img class="prodthumb" src="'+escapeHTML(safeImage(p.image_url))+'" alt="">':'<span class="prodthumb" style="display:inline-grid;place-items:center">b.</span>')+'<strong>'+escapeHTML(p.name)+'</strong></td><td>'+money(p.price)+'</td><td><span class="status '+(p.is_active?'':'off')+'">'+(p.is_active?'Live':'Hidden')+'</span></td><td><div class="admin-actions"><button class="adminbtn secondary" data-edit="'+p.id+'">Edit</button><button class="adminbtn danger" data-delete="'+p.id+'">Delete</button></div></td></tr>').join(''):'<tr><td colspan="4" style="padding:25px 10px;color:#778175">No products yet. Add your first product →</td></tr>';
  $('adminProducts').querySelectorAll('[data-edit]').forEach(b=>b.addEventListener('click',()=>editProduct(b.dataset.edit)));
  $('adminProducts').querySelectorAll('[data-delete]').forEach(b=>b.addEventListener('click',()=>deleteProduct(b.dataset.delete)));
 }
 function editProduct(id){const p=adminProducts.find(x=>x.id===id);if(!p)return;$('productId').value=p.id;$('productName').value=p.name||'';$('productPrice').value=p.price||0;$('productCategory').value=p.category||'';$('productImage').value=p.image_url||'';$('productDescription').value=p.description||'';$('productWhatsapp').value=p.whatsapp_number||'';$('productActive').checked=!!p.is_active;$('editorTitle').textContent='Edit product';$('saveProductBtn').textContent='Update product ↗';$('saveNotice').classList.add('hidden');$('productEditor').scrollIntoView({behavior:'smooth',block:'start'})}
 async function deleteProduct(id){const p=adminProducts.find(x=>x.id===id);if(!p||!confirm('Delete "'+p.name+'"? This cannot be undone.'))return;const {error}=await db.from('products').delete().eq('id',id);if(error){toast('Could not delete product: '+error.message);return}toast('Product deleted');await refreshAdminProducts()}
 $('productForm').addEventListener('submit',async e=>{
  e.preventDefault();$('saveProductBtn').disabled=true;$('saveProductBtn').textContent='Saving…';
  const id=$('productId').value;
  const payload={name:$('productName').value.trim(),price:Number($('productPrice').value),category:$('productCategory').value.trim()||'Everyday',image_url:$('productImage').value.trim()||null,description:$('productDescription').value.trim()||null,whatsapp_number:$('productWhatsapp').value.trim().replace(/\s/g,'')||null,is_active:$('productActive').checked,updated_at:new Date().toISOString()};
  if(!payload.name||!Number.isFinite(payload.price)||payload.price<0){saveNotice('Please check the product name and price.',true);$('saveProductBtn').disabled=false;$('saveProductBtn').textContent=id?'Update product ↗':'Save product ↗';return}
  const result=id?await db.from('products').update(payload).eq('id',id):await db.from('products').insert(payload);
  $('saveProductBtn').disabled=false;$('saveProductBtn').textContent=id?'Update product ↗':'Save product ↗';
  if(result.error){saveNotice('Could not save product: '+result.error.message,true);return}
  saveNotice('Saved successfully. Your storefront catalog is updated.',false);toast('Product saved ✳');resetForm();await refreshAdminProducts();
 });
 checkSession();
}
})();
